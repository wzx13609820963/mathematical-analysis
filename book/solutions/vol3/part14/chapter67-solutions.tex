\chapter{\texorpdfstring{\(L^p\)}{Lp} 空间}

\begin{solution}{lp-quotient-well-defined}
若 \(f=f'\)、\(g=g'\) 几乎处处，则
\(f+g=f'+g'\)、\(\alpha f=\alpha f'\) 几乎处处，所以运算与代表元
无关。正文的幂次估计保证和、数乘仍属于 \(\mathcal L^p\)，故商空间上的
运算封闭。又 \(|f|^p=|f'|^p\) 几乎处处，积分相同，故所定义的量
与代表元无关；只有 \(p\ge1\) 时才称它为范数。
\end{solution}

\begin{solution}{young-calculus-proof}
固定 \(b\ge0\)，令
\[
 \phi(a)=\frac{a^p}{p}-ab+\frac{b^q}{q}.
\]
则 \(\phi'(a)=a^{p-1}-b\)，唯一极小点为
\(a=b^{1/(p-1)}=b^{q-1}\)。代入得
\[
 \phi(a_{\min})
 =\frac{b^q}{p}-b^q+\frac{b^q}{q}=0.
\]
故 \(\phi(a)\ge0\)，等号恰在
\(a^{p-1}=b\)，等价于 \(a^p=b^q\)。
\end{solution}

\begin{solution}{holder-equality}
若某个范数为零，相应函数几乎处处为零，两侧都为零，恒取等号。
以下假设两个范数均为正。归一化后，Young 不等式的非负差的积分为零
当且仅当该差几乎处处为零。因此等号条件为
\[
 \frac{|f|^p}{\|f\|_p^p}
 =\frac{|g|^q}{\|g\|_q^q}
\quad\text{几乎处处}.
\]
因此非零函数的等号条件是存在 \(c>0\)，使
\(|f|^p=c|g|^q\) 几乎处处。乘积积分使用绝对值，所以 \(f,g\) 的符号
不影响 \(\|fg\|_1\) 版本；若讨论
\(|\int fg|\) 的等号，还需 \(fg\) 几乎处处同号。
\end{solution}

\begin{solution}{holder-indicator}
令 \(q=p/(p-1)\)。Hölder 给
\[
 \int_E|f|=\int|f|\mathbf1_E
 \le\|f\|_p\|\mathbf1_E\|_q
 =\|f\|_p\mu(E)^{1/q}.
\]
而 \(1/q=1-1/p\)。
\end{solution}

\begin{solution}{multilinear-holder-proof}
把所有 \(p_j=\infty\) 的因子用本质上确界提出积分号。若余下某个函数的
范数为零，结论显然；否则令
\(F_j=|f_j|/\|f_j\|_{p_j}\)。有限指数满足
\(\sum_j1/p_j=1\)，故加权算术--几何平均不等式给出
\[
 \prod_jF_j
 =\prod_j(F_j^{p_j})^{1/p_j}
 \le\sum_j\frac{F_j^{p_j}}{p_j}.
\]
积分后右侧为 \(1\)，乘回各范数即得结论。若某个 \(p_j=1\)，其余指数
只能全为 \(\infty\)；上述论证退化为
\[
 \left\|f_j\prod_{k\ne j}f_k\right\|_1
 \le\|f_j\|_1\prod_{k\ne j}\|f_k\|_\infty,
\]
没有遗漏端点情形。
\end{solution}

\begin{solution}{minkowski-integrability-gap}
凸性给
\[
 |f+g|^p\le(|f|+|g|)^p
 \le2^{p-1}(|f|^p+|g|^p).
\]
右侧可积，所以 \(f+g\in L^p\)，其范数有限。此后 Hölder 右侧和
\(\|f+g\|_p^{p-1}\) 才有有限意义；若范数为无穷，约去它不是合法代数
操作。
\end{solution}

\begin{solution}{lp-parallelogram}
逐点恒等式
\[
 |f+g|^2+|f-g|^2=2|f|^2+2|g|^2
\]
积分即得。在两点空间取 \(f=(1,0),g=(0,1)\)。若 \(p\ne2\)，则
\[
 \|f+g\|_p^2+\|f-g\|_p^2=2\cdot2^{2/p},
\qquad
 2\|f\|_p^2+2\|g\|_p^2=4,
\]
两者仅在 \(p=2\) 相等。
\end{solution}

\begin{solution}{quasi-lp-metric}
由 \(0<p<1\) 时
\(|a+b|^p\le|a|^p+|b|^p\)，积分后得到 \(d_p\) 的三角不等式；其余
度量公理直接成立。若 \((f_n)\) 为 \(d_p\)-Cauchy，选子列使
\[
 d_p(f_{n_{k+1}},f_{n_k})<2^{-k}.
\]
对非负部分和使用单调收敛定理，由其积分至多为 \(\sum_k2^{-k}\)，得
\[
 \sum_k|f_{n_{k+1}}-f_{n_k}|^p<\infty
\]
几乎处处。固定这样的点，并记
\(a_k=|f_{n_{k+1}}-f_{n_k}|\)。于是 \(a_k\to0\)，所以最终
\(a_k\le1\)，而 \(0<p<1\) 时 \(a_k\le a_k^p\)。因此
\(\sum_ka_k<\infty\)，差分级数绝对收敛，可据此定义逐点极限 \(f\)。
在上述非负级数发散的可测零集上置零，得到可测函数。由
\[
 |f-f_{n_k}|^p
 \le\sum_{j=k}^{\infty}|f_{n_{j+1}}-f_{n_j}|^p
\]
和非负部分和的单调收敛，
\[
 d_p(f_{n_k},f)
 \le\sum_{j=k}^{\infty}d_p(f_{n_{j+1}},f_{n_j})
 <\sum_{j=k}^{\infty}2^{-j}\longrightarrow0.
\]
由次可加性，
\(\int|f|^p\le d_p(f,f_{n_1})+\int|f_{n_1}|^p<\infty\)，故 \(f\in L^p\)。
给定 \(\varepsilon>0\)，先取充分大的 \(k\) 使子列误差小于
\(\varepsilon/2\)，并使 \(n_k\) 超过原列的 Cauchy 阈值，再用
\(d_p(f_n,f)\le d_p(f_n,f_{n_k})+d_p(f_{n_k},f)\)，得到整列收敛。
\end{solution}

\begin{solution}{riesz-fischer-fatou-step}
对 \(m>k\)，Minkowski 给
\[
 \|f_{n_m}-f_{n_k}\|_p
 \le\sum_{j=k}^{m-1}\|f_{n_{j+1}}-f_{n_j}\|_p
 <\sum_{j=k}^{m-1}2^{-j}.
\]
子列几乎处处趋于 \(f\)。对
\(|f_{n_m}-f_{n_k}|^p\) 使用 Fatou 引理：
\[
 \|f-f_{n_k}\|_p^p
 \le\liminf_m\|f_{n_m}-f_{n_k}\|_p^p
 \le\left(\sum_{j=k}^{\infty}2^{-j}\right)^p.
\]
取 \(p\) 次根即得结论。
\end{solution}

\begin{solution}{lp-convergence-series}
先设 \(1\le p<\infty\)。令 \(g_N=\sum_{n=1}^{N}|f_n|\)。Minkowski 给
\[
 \|g_N\|_p\le\sum_{n=1}^{N}\|f_n\|_p\le C.
\]
单调收敛得 \(g=\sum_n|f_n|\in L^p\)，所以级数几乎处处绝对收敛。其
部分和之差满足
\[
 \left\|\sum_{n=M}^{N}f_n\right\|_p
 \le\sum_{n=M}^{N}\|f_n\|_p\to0.
\]
在级数不绝对收敛的可测零集上把和定义为零，所得函数 \(s\) 可测，
且 \(|s|\le g\) 几乎处处，故 \(s\in L^p\)。
对尾和估计令 \(N\to\infty\)，Fatou 引理给
\[
 \left\|s-\sum_{n=1}^{M-1}f_n\right\|_p
 \le\sum_{n=M}^{\infty}\|f_n\|_p\longrightarrow0.
\]

当 \(0<p<1\) 且 \(\sum_n\|f_n\|_p^p<\infty\) 时，单调收敛给
\(\int\sum_n|f_n|^p<\infty\)。几乎处处有 \(\sum_n|f_n|^p<\infty\)，
故最终 \(|f_n|\le1\)，从而 \(\sum_n|f_n|<\infty\)。
同样在例外集补零定义 \(s\)，次可加性与 Fatou 给出
\[
 \int|s|^p\le\sum_n\|f_n\|_p^p,\qquad
 \int\left|s-\sum_{n=1}^{M-1}f_n\right|^p
 \le\sum_{n=M}^{\infty}\|f_n\|_p^p\longrightarrow0.
\]
为否定原条件，取实线上两两不交的单位区间 \(I_n=[n,n+1)\)，令
\(f_n=n^{-r}\mathbf1_{I_n}\)，其中 \(1<r\le1/p\)。
则 \(\sum_n\|f_n\|_p=\sum_nn^{-r}<\infty\)，而逐点和的
\(p\) 次积分为 \(\sum_nn^{-rp}=\infty\)。部分和的 \(p\) 次积分
趋于无穷，所以它们不可能在 \(L^p\) 中收敛。
\end{solution}

\begin{solution}{clarkson-special-case}
先证标量式。对任意实数 \(u,v\) 及 \(p\ge2\)，
\[
 |u|^p+|v|^p\le(u^2+v^2)^{p/2}.
\]
当右侧非零时，除以它后，两项分别为
\((u^2/(u^2+v^2))^{p/2}\) 和 \((v^2/(u^2+v^2))^{p/2}\)，
不超过对应的底数，和至多为1；零值情形直接成立。
另一方面，\(t^{2/p}\) 的凹性给出
\[
 a^2+b^2\le2^{1-2/p}(|a|^p+|b|^p)^{2/p}.
\]
令 \(u=a+b,v=a-b\)，利用 \(u^2+v^2=2(a^2+b^2)\)，得到
\[
 |a+b|^p+|a-b|^p\le2^{p-1}(|a|^p+|b|^p)
\]
逐点取 \(a=f(x),b=g(x)\) 并积分即得题设不等式。
当 \(\|f\|_p,\|g\|_p\le1\) 时，两边除以 \(2^p\)，可得
\[
 \left\|\frac{f+g}{2}\right\|_p^p+
 \left\|\frac{f-g}{2}\right\|_p^p\le1.
\]
若 \(\|f-g\|_p\ge\varepsilon\)，第二项至少为 \((\varepsilon/2)^p\)。
移项并取 \(p\) 次根，即得所求估计。因而可取一致凸性模的下界
\(\delta(\varepsilon)=1-(1-(\varepsilon/2)^p)^{1/p}>0\)，
它不依赖具体的 \(f,g\)。
\end{solution}

\begin{solution}{lp-completeness-template}
模板是：从 Cauchy 列抽子列 \(x_{n_k}\)，使
\(\sum_k\|x_{n_{k+1}}-x_{n_k}\|<\infty\)；若空间能把绝对可求和差分
转化为一个候选极限，再用尾和估计和 Cauchy 性恢复整列收敛，即得完备。

在 \(C_b(X)\) 中，令
\(\|f_{n_{k+1}}-f_{n_k}\|_\infty<2^{-k}\)。Weierstrass 判别法说明差分
级数一致收敛，其和与 \(f_{n_1}\) 之和是有界连续函数 \(f\)。尾和估计给
\(\|f_{n_k}-f\|_\infty\to0\)，原列 Cauchy 因而也收敛于 \(f\)。
\end{solution}
